Snippet Verifikasi Token untuk Aplikasi Baru Anda

Copy kode JavaScript murni ini ke aplikasi software apa pun yang Anda bangun.

Untuk memvalidasi token yang dihasilkan oleh generator ini di aplikasi baru Anda (misal: Toko Retail, Laundry, Apotek, atau WebApp SaaS), cukup tempelkan fungsi JavaScript berikut ke dalam file script aplikasi Anda: 

// ============================================================
// RASH UNIVERSAL LICENSE VERIFIER (Drop-in JS Snippet)
// ============================================================
function verifyUniversalLicense(rawToken, expectedAppCode = "RESTO") {
  const token = (rawToken || "").trim().toUpperCase();
  if (!token) return { valid: false, message: "Token lisensi tidak boleh kosong!" };

  // Format standar: RASH-[APP]-PRO-... atau RASH-PRO-...
  const isValidFormat = token.startsWith(`RASH-${expectedAppCode}-PRO-`) || 
                        token.startsWith("RASH-PRO-") || 
                        token.startsWith("RASH-APP-PRO-") ||
                        token.includes("LIFETIME") || 
                        token.includes("VIP");

  if (!isValidFormat) {
    return { valid: false, message: `Token tidak sesuai untuk aplikasi ${expectedAppCode}!` };
  }

  let expiry = "31 Desember 2028";
  if (token.includes("LIFETIME") || token.includes("VIP")) {
    expiry = "Permanen / Tanpa Batas (Lifetime)";
  } else if (token.includes("2027")) {
    expiry = "31 Desember 2027";
  }

  const licenseData = {
    isPro: true,
    token: token,
    appCode: expectedAppCode,
    expiryDateStr: expiry,
    activatedAt: new Date().toISOString()
  };

  localStorage.setItem("app_saas_license", JSON.stringify(licenseData));
  return { valid: true, message: `Lisensi Pro Berhasil Aktif (${expiry})!`, data: licenseData };
}
